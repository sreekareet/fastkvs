cmake_minimum_required(VERSION 3.10)
project(FastKV)

set(CMAKE_CXX_STANDARD 17)

add_executable(fastkv
    src/main.cpp
    src/kvstore.cpp
    src/thread_pool.cpp
    src/lru_cache.cpp
    src/persistence.cpp
)

add_executable(lru_test
    test/lru_evict_test.cpp
    src/kvstore.cpp
    src/thread_pool.cpp
    src/lru_cache.cpp
    src/persistence.cpp
)
add_executable(benchmark_single
     test/benchmark_single.cpp
     src/kvstore.cpp
     src/thread_pool.cpp
     src/lru_cache.cpp
     src/persistence.cpp
 )
add_executable(benchmark_multi
      test/benchmark_multi.cpp
      src/kvstore.cpp
      src/thread_pool.cpp
      src/lru_cache.cpp
      src/persistence.cpp
  )

target_compile_options(fastkv PRIVATE -Wall -Wextra -Wpedantic)

find_package(Threads REQUIRED)
target_link_libraries(fastkv PRIVATE Threads::Threads)
